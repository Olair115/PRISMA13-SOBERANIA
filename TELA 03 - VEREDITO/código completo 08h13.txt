    function loadDataFromStorage() {
        try {
            const savedHistory = localStorage.getItem('appnutri_evaluations');
            const savedCurrent = localStorage.getItem('appnutri_current');
            const savedLicense = localStorage.getItem('appnutri_license');

            localStorage.clear();
        }

        if (screenId === 'bloco9' && !hasActiveLicense) {
            alert("O histórico de evolução é um recurso exclusivo para assinantes. Ative um plano para desbloquear!");
            renderScreen('bloco6');
            return;
        }

        document.querySelectorAll('main > div').forEach(div => {
            div.classList.remove('visible');
            div.classList.add('hidden');
        });
        const container = document.getElementById(screenId);
        if (!container) {
            console.error("Tela não encontrada:", screenId);
            return;
        }
        container.classList.remove('hidden');
        
        setTimeout(() => { container.classList.add('visible'); }, 10);

        switch (screenId) {
            case 'bloco0': renderBloco0(); break;
            case 'bloco1': renderBloco1(); break;
            case 'bloco2': renderBloco2(); break;
            case 'bloco2_interpretacao': renderBloco2Interpretacao(); break;
            case 'bloco3': renderBloco3(); break;
            case 'bloco4': renderBloco4(); break;
            case 'bloco4_parcial': renderBloco4Parcial(); break;
            case 'bloco5': renderBloco5(); break;
            case 'bloco6': renderBloco6(); break;
            case 'bloco7': renderBloco7(); break;
            case 'bloco8': renderBloco8(); break;
            case 'bloco9': renderBloco9(); break;
        }
        window.scrollTo(0, 0);
    }

    function handleDataSubmit() {
        const errorEl = document.getElementById('formError');
        errorEl.textContent = '';
        try {
            currentEvaluation = { gender: document.getElementById('gender').value, age: parseInt(document.getElementById('age').value), height: parseFloat(document.getElementById('height').value), weight: parseFloat(document.getElementById('weight').value), work_activity: document.getElementById('work_activity').value, physical_activity: document.getElementById('physical_activity').value, current_diseases: document.getElementById('current_diseases').value, past_diseases: document.getElementById('past_diseases').value };
            if (isNaN(currentEvaluation.age) || isNaN(currentEvaluation.height) || isNaN(currentEvaluation.weight) || currentEvaluation.age <= 0 || currentEvaluation.height <= 0 || currentEvaluation.weight <= 0) { throw new Error('Por favor, preencha idade, altura e peso corretamente.'); }
            currentEvaluation.bmi = currentEvaluation.weight / (currentEvaluation.height * currentEvaluation.height);
            currentEvaluation.data = new Date().toISOString();
            currentEvaluation.areas = {};
            currentAreaIndex = 0;
            saveDataToStorage();
            renderScreen('bloco2_interpretacao');
        } catch (error) { errorEl.textContent = error.message; }
    }

    function renderBloco3() {
        const container = document.getElementById('bloco3');
        const currentArea = AREAS_DA_VIDA[currentAreaIndex];
        const progressPercent = ((currentAreaIndex) / AREAS_DA_VIDA.length) * 100;
        const questionsHTML = currentArea.questoes.map((q, index) => {

            return `<div class="question-block"><p class="question-text">${index + 1}. ${q}</p><div class="rating-slider"><div class="slider-container"><input type="range" min="0" max="99" value="${savedValue}" class="slider" id="slider-${index}" oninput="updateSliderValue(this.value, ${index})" style="--value: ${savedValue}; --max: 99;"><div class="slider-value" id="sliderValue-${index}">${String(savedValue).padStart(2, '0')}</div></div><div class="slider-labels"><span>00 - Totalmente Insatisfeito</span><span>99 - Plenamente Satisfeito</span></div></div></div>`;
        }).join('');
        container.innerHTML = `<h2 class="form-title">Vagão 3: Autoavaliação</h2><div class="progress-bar"><div class="progress-fill" style="width: ${progressPercent}%"></div></div><div class="progress-text">Progresso da Viagem: Dimensão ${currentAreaIndex + 1} de ${AREAS_DA_VIDA.length}</div><div class="area-container"><div class="area-title"><span class="area-icon">${currentArea.icon}</span><span>${currentArea.nome}</span></div>${questionsHTML}</div><div class="button-group"><button class="button button-secondary" onclick="goBackInEvaluation()">← Vagão Anterior</button><button class="button button-primary" onclick="saveDimensionRatings()">${currentAreaIndex === AREAS_DA_VIDA.length - 1 ? 'Finalizar Avaliação' : 'Próximo Vagão →'}</button></div>`;
    }
    
    function updateSliderValue(value, index) {
        document.getElementById(`sliderValue-${index}`).textContent = String(value).padStart(2, '0');
        const sliderEl = document.getElementById(`slider-${index}`);
        sliderEl.style.setProperty('--value', value);
        sliderEl.style.setProperty('--max', sliderEl.max);
    }

    function saveDimensionRatings() {
        const currentArea = AREAS_DA_VIDA[currentAreaIndex];
        const ratings = Array.from({ length: currentArea.questoes.length }, (_, i) => parseInt(document.getElementById(`slider-${i}`).value));
        if (!currentEvaluation.areas) currentEvaluation.areas = {};
        currentEvaluation.areas[currentArea.nome] = ratings;
        currentAreaIndex++;
        if (currentAreaIndex >= AREAS_DA_VIDA.length) { finishEvaluation(); } else { saveDataToStorage(); renderScreen('bloco3'); }
    }

    function goBackInEvaluation() {
        if (currentAreaIndex > 0) { currentAreaIndex--; renderScreen('bloco3'); } else { renderScreen('bloco2_interpretacao'); }
    }

    function finishEvaluation() {
        currentEvaluation.medias = {};
        Object.keys(currentEvaluation.areas).forEach(areaKey => {
            const areaRatings = currentEvaluation.areas[areaKey];
            currentEvaluation.medias[areaKey] = areaRatings.reduce((a, b) => a + b, 0) / areaRatings.length;
        });
        const medias = Object.values(currentEvaluation.medias);
        currentEvaluation.mediaGeral = medias.reduce((a, b) => a + b, 0) / medias.length;
        
        evaluationHistory.push({...currentEvaluation});
        saveDataToStorage();
        
        const isPaidUser = userLicense.type !== 'none' && userLicense.type !== 'free';
        if (isPaidUser) {
            renderScreen('bloco4');
        } else {
            renderScreen('bloco4_parcial');
        }
    }
