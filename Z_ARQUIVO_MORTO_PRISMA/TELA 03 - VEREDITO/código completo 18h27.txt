    function loadDataFromStorage() {
        try {
            const savedHistory = localStorage.getItem('appnutri_evaluations');
            const savedCurrent = localStorage.getItem('appnutri_current');
            const savedLicense = localStorage.getItem('appnutri_license');

    function handleDataSubmit() {
        const errorEl = document.getElementById('formError');
        errorEl.textContent = '';

        try {
            currentEvaluation = {
                gender: document.getElementById('gender').value,
                age: parseInt(document.getElementById('age').value),
                height: parseFloat(document.getElementById('height').value),
                weight: parseFloat(document.getElementById('weight').value),
                work_activity: document.getElementById('work_activity').value,
                physical_activity: document.getElementById('physical_activity').value,
                current_diseases: document.getElementById('current_diseases').value,
                past_diseases: document.getElementById('past_diseases').value
            };
            if (isNaN(currentEvaluation.age) || isNaN(currentEvaluation.height) || isNaN(currentEvaluation.weight) || currentEvaluation.age <= 0 || currentEvaluation.height <= 0 || currentEvaluation.weight <= 0) {
                throw new Error('Por favor, preencha idade, altura e peso corretamente.');
            }

            currentEvaluation.bmi = currentEvaluation.weight / (currentEvaluation.height * currentEvaluation.height);
            currentEvaluation.data = new Date().toISOString();
            currentEvaluation.areas = {};
            currentAreaIndex = 0;
            
            saveDataToStorage();
            renderScreen('bloco2_interpretacao');
        } catch (error) {
            errorEl.textContent = error.message;
        }
    }

                <button class="button button-primary" onclick="saveDimensionRatings()">${currentAreaIndex === AREAS_DA_VIDA.length - 1 ? 'Finalizar Avaliação' : 'Próximo Vagão →'}</button>
            </div>
        `;
    }
    
    function updateSliderValue(value, index) {
        document.getElementById(`sliderValue-${index}`).textContent = String(value).padStart(2, '0');
        
        // Adição para atualizar o estilo da trilha
        const sliderEl = document.getElementById(`slider-${index}`);
        sliderEl.style.setProperty('--value', value);
        sliderEl.style.setProperty('--max', sliderEl.max);
    }

    function saveDimensionRatings() {
        const currentArea = AREAS_DA_VIDA[currentAreaIndex];
        const ratings = [];
        for (let i = 0; i < currentArea.questoes.length; i++) {
            const rating = parseInt(document.getElementById(`slider-${i}`).value);
            ratings.push(rating);
        }
        
        if (!currentEvaluation.areas) currentEvaluation.areas = {};
        currentEvaluation.areas[currentArea.nome] = ratings;
        
        currentAreaIndex++;
        
        if (currentAreaIndex >= AREAS_DA_VIDA.length) {
            finishEvaluation();
        } else {
            saveDataToStorage();
            renderScreen('bloco3');
        }
    }

    function goBackInEvaluation() {
        if (currentAreaIndex > 0) {
            currentAreaIndex--;
            renderScreen('bloco3');
        } else {
            renderScreen('bloco2_interpretacao');
        }
    }

    function finishEvaluation() {
        currentEvaluation.medias = {};
        Object.keys(currentEvaluation.areas).forEach(areaKey => {
            const areaRatings = currentEvaluation.areas[areaKey];
            currentEvaluation.medias[areaKey] = areaRatings.reduce((a, b) => a + b, 0) / areaRatings.length;
        });
        const medias = Object.values(currentEvaluation.medias);
        currentEvaluation.mediaGeral = medias.reduce((a, b) => a + b, 0) / medias.length;
        
        evaluationHistory.push({...currentEvaluation});
        currentAreaIndex = 0;
        // Reset for a new evaluation
        saveDataToStorage();
        renderScreen('bloco4');
    }

    function renderBloco9() {
        const container = document.getElementById('bloco9');

            return;
        }
        
        let variationStats = {};
        let daysElapsed = 0;
        
        if (evaluationHistory.length >= 2) {
            const latest = evaluationHistory[evaluationHistory.length - 1];
            const previous = evaluationHistory[evaluationHistory.length - 2];
            daysElapsed = Math.floor((new Date(latest.data) - new Date(previous.data)) / (1000 * 60 * 60 * 24));
            variationStats = {
                mediaGeral: { current: latest.mediaGeral, previous: previous.mediaGeral, variation: ((latest.mediaGeral - previous.mediaGeral) / (previous.mediaGeral || 1)) * 100 },
                bmi: { current: latest.bmi, previous: previous.bmi, variation: ((latest.bmi - previous.bmi) / (previous.bmi || 1)) * 100 }
            };
        }
        
        container.innerHTML = `
            <h2 class="form-title">Vagão 9: Sua Evolução</h2>
            ${evaluationHistory.length >= 2 ? `<div class="interpretation-box"><h4>📈 Variação desde a última viagem</h4><div class="variation-stats"><div class="stat-card"><div class="stat-value ${variationStats.mediaGeral.variation >= 0 ? 'positive' : 'negative'}">${variationStats.mediaGeral.variation >= 0 ? '+' : ''}${variationStats.mediaGeral.variation.toFixed(1)}%</div><div class="stat-label">Variação da Pontuação</div><div style="font-size: 0.8rem; color: var(--cor-texto-terciario);">${(variationStats.mediaGeral.previous || 0).toFixed(0)} → ${(variationStats.mediaGeral.current || 0).toFixed(0)}</div></div><div class="stat-card"><div class="stat-value ${variationStats.bmi.variation <= 0 ? 'positive' : 'negative'}">${variationStats.bmi.variation > 0 ? '+' : ''}${variationStats.bmi.variation.toFixed(1)}%</div><div class="stat-label">Variação do IMC</div><div style="font-size: 0.8rem; color: var(--cor-texto-terciario);">${(variationStats.bmi.previous || 0).toFixed(1)} → ${(variationStats.bmi.current || 0).toFixed(1)}</div></div><div class="stat-card"><div class="stat-value neutral">${daysElapsed}</div><div class="stat-label">Dias Entre Viagens</div></div></div></div>` : ''}

            <div class="chart-container"><div class="chart-title">Evolução do IMC</div><div class="chart-wrapper"><canvas id="bmiChart"></canvas></div></div>

    function renderCharts() {
        const labels = evaluationHistory.map(e => new Date(e.data).toLocaleDateString('pt-BR'));
        const baseOptions = { responsive: true, maintainAspectRatio: false, plugins: { legend: { labels: { color: '#E0E0E0' } } }, scales: { x: { ticks: { color: '#BDBDBD' }, grid: { color: '#333' } } } };
        new Chart(document.getElementById('generalChart').getContext('2d'), { type: 'line', data: { labels, datasets: [{ label: 'Pontuação Geral', data: evaluationHistory.map(e => e.mediaGeral), borderColor: '#8A63D2', backgroundColor: 'rgba(138, 99, 210, 0.1)', borderWidth: 3, fill: true, tension: 0.4 }] }, options: {...baseOptions, scales: {...baseOptions.scales, y: { beginAtZero: true, max: 100, ticks: { color: '#BDBDBD' }, grid: { color: '#333' } } } } });
        new Chart(document.getElementById('bmiChart').getContext('2d'), { type: 'line', data: { labels, datasets: [{ label: 'IMC', data: evaluationHistory.map(e => e.bmi), borderColor: '#10B981', backgroundColor: 'rgba(16, 185, 129, 0.1)', borderWidth: 3, fill: true, tension: 0.4 }] }, options: {...baseOptions, scales: {...baseOptions.scales, y: { beginAtZero: false, ticks: { color: '#BDBDBD' }, grid: { color: '#333' } } } } });
        const colors = ['#10B981', '#EF4444', '#3B82F6', '#8B5CF6', '#F59E0B', '#6366F1', '#EC4899', '#14B8A6'];
        const areaDatasets = AREAS_DA_VIDA.map((area, index) => ({ label: area.nome, data: evaluationHistory.map(e => (e.medias && e.medias[area.nome]) || 0), borderColor: colors[index % colors.length], backgroundColor: 'transparent', borderWidth: 2, tension: 0.4 }));
        new Chart(document.getElementById('areasChart').getContext('2d'), { type: 'line', data: { labels, datasets: areaDatasets }, options: {...baseOptions, scales: {...baseOptions.scales, y: { beginAtZero: true, max: 100, ticks: { color: '#BDBDBD' }, grid: { color: '#333' } } } } });
    }

    function getBmiInterpretation(evalData) {
        const { age, bmi, physical_activity } = evalData;
        let result = { category: "", text: "", warning: "", color: "var(--cor-alerta)", icon: "🤔" };
        if (age < 20) {
            result.category = "Avaliação para Jovens";
            result.text = "Nesta fase, o IMC é avaliado por curvas de percentil (ideal entre 5 e 85). Uma avaliação profissional é crucial para interpretar corretamente o resultado e o desenvolvimento corporal.";
        } else if (age >= 60) {
            if (bmi < 22) { 
                result.category = "Atenção à Nutrição (Idosos)";
                result.text = "IMC abaixo de 22 pode indicar risco de desnutrição e perda de massa muscular (sarcopenia), o que é um ponto de atenção importante para a saúde e autonomia.";
                result.color = "var(--cor-erro)";
                result.icon = "⚠️";
            } else if (bmi <= 27) { 
                result.category = "Faixa Saudável para a Maturidade";
                result.text = "Excelente! Manter o IMC entre 22 e 27 está associado a uma maior reserva funcional e a um envelhecimento mais saudável.";
                result.color = "var(--cor-sucesso)";
                result.icon = "✅";
            } else { 
                result.category = "Ponto de Atenção (Idosos)";
                result.text = "IMC acima de 27 pode aumentar o risco de condições metabólicas e sobrecarga nas articulações, sendo importante um acompanhamento.";
            }
        } else {
            if (bmi < 18.5) { 
                result.category = "Atenção à Nutrição (Adultos)";
                result.text = "IMC abaixo de 18.5 pode indicar necessidade de maior aporte nutricional para garantir energia, imunidade e saúde óssea.";
                result.color = "var(--cor-erro)";
                result.icon = "⚠️";
            } else if (bmi < 25) { 
                result.category = "Faixa de Normalidade (Adultos)";
                result.text = "Parabéns! Seu IMC está na faixa ideal, o que é um excelente indicador de saúde e está associado a menores riscos de doenças crônicas.";
                result.color = "var(--cor-sucesso)";
                result.icon = "✅";
            } else if (bmi < 30) { 
                result.category = "Ponto de Atenção: Sobrepeso";
                result.text = "Seu IMC indica sobrepeso. Este é um momento oportuno para reavaliar hábitos alimentares e de atividade física para prevenir futuras complicações.";
            } else { 
                result.category = "Cuidado Ampliado: Obesidade";
                result.text = "Seu IMC está na faixa de obesidade. Esta condição aumenta significativamente o risco de diversas doenças crônicas e requer atenção e cuidado profissional.";
                result.color = "var(--cor-erro)";
                result.icon = "🚨";
            }
        }
        
        const activityNormalized = (physical_activity || "").toLowerCase();
        if (bmi >= 25 && (activityNormalized.includes('musculação') || activityNormalized.includes('intenso') || activityNormalized.includes('atleta'))) {
            result.warning = '<strong>Nota (Paradoxo de Holyfield):</strong> Dado seu nível de atividade física, um IMC elevado pode refletir maior massa muscular, e não necessariamente excesso de gordura. Uma avaliação de composição corporal (bioimpedância, dobras cutâneas) é recomendada para um diagnóstico preciso.';
        }
        return result;
    }
