# MANUAL DE IMPLANTAÇÃO - PRISMA12 (Módulo de Acessos e Nutrição)

Este documento contém a arquitetura de bloqueio de rotas, trial de 7 dias e integração com Firebase para o aplicativo PRISMA12. Siga a ordem exata abaixo para a configuração.

### PARTE 1: Configuração no Console do Firebase

**1. Habilitar Autenticação Anônima**
* Acesse o Firebase Console.
* Vá em **Authentication** -> aba **Sign-in method**.
* Ative a opção **Anonymous** (Anônimo) e salve.

**2. Publicar as Regras do Firestore**
* Vá em **Firestore Database** -> aba **Rules**.
* Apague o conteúdo atual, cole o bloco abaixo e clique em **Publish**:

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function isMe(uid) { return signedIn() && request.auth.uid == uid; }

    match /users/{uid} {
      allow read: if isMe(uid);
      allow create: if isMe(uid);

      // Usuário NÃO pode alterar entitlements (trial/premium)
      allow update: if isMe(uid)
        && !( 'premium' in request.resource.data
           || 'trialEndsAt' in request.resource.data
           || 'premiumUntil' in request.resource.data );
    }

    // Permite histórico por trilha
    match /users/{uid}/{colId}/{docId} {
      allow read, write: if isMe(uid) && colId.matches('assessments_.*');
    }

    match /hotmart_events/{docId} {
      allow read, write: if false;
    }
  }
}
```

**2. lib/core/auth_controller.dart**
```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';

class AuthController {
  final FirebaseAuth _auth;
  final FirebaseFirestore _db;

  AuthController({FirebaseAuth? auth, FirebaseFirestore? db})
      : _auth = auth ?? FirebaseAuth.instance,
        _db = db ?? FirebaseFirestore.instance;

  Future ensureSignedInAndBootstrapTrial() async {
    if (_auth.currentUser == null) {
      await _auth.signInAnonymously();
    }

    final user = _auth.currentUser;
    if (user == null) return;

    final docRef = _db.collection('users').doc(user.uid);
    final snap = await docRef.get();

    if (!snap.exists) {
      await docRef.set({
        'createdAt': FieldValue.serverTimestamp(),
        'trialEndsAt': Timestamp.fromDate(DateTime.now().add(const Duration(days: 7))),
        'premium': false,
      }, SetOptions(merge: true));
    }
  }
}
```

**3. lib/core/entitlements_provider.dart**
```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

class Entitlements {
  final bool premiumActive;
  final bool trialActive;
  final DateTime? trialEndsAt;

  const Entitlements({
    required this.premiumActive,
    required this.trialActive,
    this.trialEndsAt,
  });

  bool get hasAccess => premiumActive || trialActive;
  static const none = Entitlements(premiumActive: false, trialActive: false);
}

final entitlementsProvider = StreamProvider((ref) {
  final user = FirebaseAuth.instance.currentUser;
  if (user == null) return Stream.value(Entitlements.none);

  return FirebaseFirestore.instance
      .collection('users')
      .doc(user.uid)
      .snapshots()
      .map((snap) {
    if (!snap.exists) return Entitlements.none;

    final data = snap.data() ?? {};
    final premium = (data['premium'] as bool?) ?? false;
    final ts = data['trialEndsAt'] as Timestamp?;
    final ends = ts?.toDate();
    final trial = ends != null && DateTime.now().isBefore(ends);

    return Entitlements(
      premiumActive: premium,
      trialActive: trial,
      trialEndsAt: ends,
    );
  });
});
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../entitlements_provider.dart';
import '../../features/nutrition_self_assessment/presentation/screens/paywall_screen.dart';

class PremiumGuard extends ConsumerWidget {
  final Widget child;
  final String? requestedLocation;

  const PremiumGuard({
    super.key,
    required this.child,
    required this.requestedLocation,
  });

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final entAsync = ref.watch(entitlementsProvider);

    return entAsync.when(
      data: (ent) {
        if (ent.hasAccess) return child;
        return PaywallScreen(requestedLocation: _safe(requestedLocation));
      },
      loading: () => const Scaffold(body: Center(child: CircularProgressIndicator())),
      error: (_, __) => PaywallScreen(requestedLocation: _safe(requestedLocation)),
    );
  }

  String _safe(String? loc) {
    final v = (loc ?? '').trim();
    if (v.isEmpty) return '/nutricao';
    if (!v.startsWith('/')) return '/nutricao';
    if (v.contains('/paywall')) return '/nutricao';
    return v;
  }
}
```

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';

class PaywallScreen extends ConsumerStatefulWidget {
  final String? requestedLocation;
  const PaywallScreen({super.key, this.requestedLocation});

  @override
  ConsumerState createState() => _PaywallScreenState();
}

class _PaywallScreenState extends ConsumerState {
  bool _redirected = false;

  @override
  void initState() {
    super.initState();
    ref.listenManual>(entitlementsProvider, (prev, next) {
      final ent = next.asData?.value;
      if (ent == null || _redirected) return;

      if (ent.hasAccess) {
        _redirected = true;
        final target = _safeTarget(widget.requestedLocation);
        WidgetsBinding.instance.addPostFrameCallback((_) {
          if (mounted) context.go(target);
        });
      }
    });
  }

  String _safeTarget(String? loc) {
    final v = (loc ?? '').trim();
    if (v.isEmpty) return '/nutricao';
    if (!v.startsWith('/')) return '/nutricao';
    if (v.contains('/paywall')) return '/nutricao';
    return v;
  }

  String _featureName(String? loc) {
    final v = loc ?? '';
    if (v.contains('historico')) return 'seu Histórico';
    if (v.contains('comparar')) return 'a Comparação';
    if (v.contains('plano-detalhado')) return 'seu Plano Detalhado';
    if (v.contains('exportar-pdf')) return 'a Exportação em PDF';
    if (v.contains('calibrador')) return 'o Calibrador';
    return 'este recurso';
  }

  @override
  Widget build(BuildContext context) {
    final feature = _featureName(widget.requestedLocation);

    return Scaffold(
      appBar: AppBar(title: const Text('Premium')),
      body: Padding(
        padding: const EdgeInsets.all(24),
```

**7. lib/main.dart**
```dart
import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import 'firebase_options.dart';
import 'core/auth_controller.dart';
import 'core/routing/app_router.dart';

Future main() async {
  WidgetsFlutterBinding.ensureInitialized();

  await Firebase.initializeApp(
    options: DefaultFirebaseOptions.currentPlatform,
  );

  final auth = AuthController();
  await auth.ensureSignedInAndBootstrapTrial();

  runApp(const ProviderScope(child: AppRoot()));
}

class AppRoot extends StatelessWidget {
  const AppRoot({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp.router(
      debugShowCheckedModeBanner: false,
      title: 'PRISMA12',
      theme: ThemeData(
        colorSchemeSeed: Colors.blue,
        scaffoldBackgroundColor: Colors.grey[50],
        useMaterial3: true,
      ),
      routerConfig: appRouter,
    );
  }
}
```
