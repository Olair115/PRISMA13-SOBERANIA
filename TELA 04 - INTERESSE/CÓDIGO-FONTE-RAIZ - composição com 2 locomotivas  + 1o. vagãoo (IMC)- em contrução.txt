    function handleDataSubmit() {
        const age = parseInt(document.getElementById('age').value);
        const height = parseFloat(document.getElementById('height').value);
        const weight = parseFloat(document.getElementById('weight').value);
        const activity = document.getElementById('activity').value;
        const errorEl = document.getElementById('formError');
        
        errorEl.textContent = '';
        if (isNaN(age) || isNaN(height) || isNaN(weight) || age <= 0 || height <= 0 || weight <= 0) {
            errorEl.textContent = 'Por favor, preencha todos os campos corretamente.';
            return;
        }
        
        const bmi = weight / (height * height);
        const interpretationData = getBmiInterpretation(age, bmi, activity);
        
        renderInterpretationScreen(age, bmi, interpretationData);
    }
    
    function getBmiInterpretation(age, bmi, activity) {
        let result = { category: "", text: "", warning: "" };
        if (age < 20) {
            result.category = "Avaliação para Jovens (até 19 anos)";
            result.text = "Nesta fase, o IMC é avaliado por curvas de percentil (ideal entre 5 e 85). Uma avaliação profissional é crucial.";
        } else if (age >= 60) {
            if (bmi < 22) { result.category = "Atenção à Nutrição (Idosos)"; result.text = "IMC abaixo de 22 pode indicar risco de desnutrição e perda de massa muscular (sarcopenia)."; }
            else if (bmi <= 27) { result.category = "Faixa Saudável para a Maturidade"; result.text = "Manter o IMC entre 22 e 27 está associado a uma maior reserva funcional."; }
            else { result.category = "Ponto de Atenção (Idosos)"; result.text = "IMC acima de 27 pode aumentar o risco de condições metabólicas."; }
        } else {
            if (bmi < 18.5) { result.category = "Atenção à Nutrição (Adultos)"; result.text = "IMC abaixo de 18.5 pode indicar necessidade de maior aporte nutricional."; }
            else if (bmi < 25) { result.category = "Faixa de Normalidade (Adultos)"; result.text = "Excelente! Seu IMC está na faixa ideal, associada a menores riscos de doenças crônicas."; }
            else if (bmi < 30) { result.category = "Ponto de Atenção: Sobrepeso"; result.text = "Seu IMC indica sobrepeso, uma oportunidade para reavaliar hábitos."; }
            else { result.category = "Cuidado Ampliado: Obesidade"; result.text = "Seu IMC está na faixa de obesidade, o que requer atenção e cuidado."; }
        }
        if (bmi >= 25 && activity === 'intenso') {
            result.warning = '<strong>Nota (Paradoxo de Holyfield):</strong> Como seu nível de atividade é intenso, um IMC elevado pode refletir maior massa muscular, e não excesso de gordura. Recomenda-se uma avaliação de composição corporal.';
        }
        return result;
    }
