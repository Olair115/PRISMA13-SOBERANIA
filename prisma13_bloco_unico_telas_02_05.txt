/* PRISMA13 · patch da TELA01
   Entrada automática: ao validar identificação + senha, salta para /tela02/index.html.
   IDs aceitos: #login ou #identificacao para login; #senha ou #password para senha.
*/
(() => {
  const CONFIG = {
    senhaPortal: '13',
    proximaTela: '/tela02/index.html',
    minimoLogin: 2
  };

  const login = document.querySelector('#login, #identificacao, input[type="text"]');
  const senha = document.querySelector('#senha, #password, input[type="password"]');
  const status = document.querySelector('#status-login, .portal-status, [data-status="login"]');
  let travessia = false;

  function ouro(texto) {
    if (status) status.textContent = texto;
  }

  function validarPortal() {
    if (!login || !senha || travessia) return;
    const nome = login.value.trim();
    const chave = senha.value.trim();
    const selado = nome.length >= CONFIG.minimoLogin && chave === CONFIG.senhaPortal;

    localStorage.setItem('prisma_login', nome);

    if (selado) {
      travessia = true;
      ouro('Entrada validada. Travessia iniciada.');
      document.body.classList.add('entrada-validada');
      setTimeout(() => location.replace(CONFIG.proximaTela), 180);
    } else {
      ouro('Presença em reconhecimento.');
    }
  }

  [login, senha].forEach(campo => {
    if (campo) {
      campo.addEventListener('input', validarPortal);
      campo.addEventListener('change', validarPortal);
    }
  });

  validarPortal();
})();

(function kosmos(){
  const c=document.getElementById('kosmos');
  if(!c) return;
  const ctx=c.getContext('2d',{alpha:true});
  let w,h,stars=[];
  function reset(){
    w=c.width=innerWidth*devicePixelRatio;
    h=c.height=innerHeight*devicePixelRatio;
    stars=Array.from({length:Math.floor(Math.min(420, innerWidth*.28))},()=>({
      x:Math.random()*w,y:Math.random()*h,r:(Math.random()*1.15+.25)*devicePixelRatio,
      a:Math.random()*.65+.18,s:Math.random()*.28+.04
    }));
  }
  addEventListener('resize',reset,{passive:true}); reset();
  function draw(t){
    ctx.clearRect(0,0,w,h);
    stars.forEach(p=>{
      const glow=p.a+Math.sin(t*.001*p.s+p.x)*.18;
      ctx.beginPath(); ctx.fillStyle=`rgba(255,245,185,${Math.max(.08,glow)})`;
      ctx.arc(p.x,p.y,p.r,0,Math.PI*2); ctx.fill();
    });
    requestAnimationFrame(draw);
  }
  requestAnimationFrame(draw);

  function draw(t){

    });

  }
